Demos¶

Nine standalone runnable scripts in demos/ showcase every layer of geometor.pappus—spanning both the projective Pappus–Steiner–Pascal canon and Books III, IV, V, and VIII of the Synagoge—with formatted Rich terminal tables and 100% exact SymPy coordinates.

1. demos/pappus_hexagon.py — Synagoge VII.138 & VII.139¶

Constructs both the oblique intersecting-carrier hexagon (Prop. 139) and the parallel-carrier hexagon (Prop. 138), verifies the universal 8-variable polynomial proof, and computes the exact signed areas of the three homologous square / circle / diamond triangles.

uv run --no-sync python demos/pappus_hexagon.py
        
"""Demonstration 1: Pappus's Hexagon Theorem (Synagoge VII.138 & VII.139) and Type Triangles.

        Constructs both the general intersecting-carrier case (Prop. 139) and the parallel
        carrier case (Prop. 138), verifies the universal symbolic polynomial proof over
        eight free indeterminates, and revives the homologous ``square`` / ``circle`` /
        ``diamond`` triangles from GEOMETOR's ``pappus-3.py`` exploration.
        """
        from __future__ import annotations

        import sys
        from pathlib import Path

        _SRC_DIR = Path(__file__).resolve().parents[1] / "src"
        if str(_SRC_DIR) not in sys.path:
            sys.path.insert(0, str(_SRC_DIR))

        from rich.console import Console  # noqa: E402
        from rich.table import Table  # noqa: E402

        from geometor.pappus import (  # noqa: E402
            construct_pappus_hexagon,
            line_spread,
            quadrance,
            verify_pappus_symbolic,
        )


        def run_demo() -> None:
            """Execute the Pappus Hexagon demonstration and print exact symbolic tables."""
            console = Console()

            # 1. General projective case (Synagoge Book VII, Prop. 139)
            general = construct_pappus_hexagon(
                preset="hexagon",
                perm=(0, 1, 2),
                include_hexagon=True,
                include_type_triangles=True,
            )
            table_gen = Table(title="Synagoge VII.139 — General Intersecting Carriers")
            table_gen.add_column("Element", style="cyan bold")
            table_gen.add_column("Exact Symbolic Expression", style="green")
            table_gen.add_row("Line a", f"{general.line_a.equation()} = 0")
            table_gen.add_row("Line b", f"{general.line_b.equation()} = 0")
            table_gen.add_row("Carrier Spread s(a, b)", str(line_spread(general.line_a, general.line_b)))
            if general.apex is not None:
                table_gen.add_row("Carrier Apex O = a ∩ b", f"({general.apex.x}, {general.apex.y})")
            for m_id, m_pt in general.meets.items():
                if m_pt is not None:
                    table_gen.add_row(f"Meet {m_id}", f"({m_pt.x}, {m_pt.y})")
            table_gen.add_row("Pappus Line", f"{general.pappus_line.equation()} = 0")
            table_gen.add_row("Collinear?", str(general.is_collinear))
            console.print(table_gen)

            # 2. Parallel carrier case (Synagoge Book VII, Prop. 138)
            parallel = construct_pappus_hexagon(preset="parallel", perm=(0, 1, 2))
            table_par = Table(title="Synagoge VII.138 — Parallel Carriers (s(a, b) = 0)")
            table_par.add_column("Element", style="cyan bold")
            table_par.add_column("Exact Symbolic Expression", style="green")
            table_par.add_row("Carrier Spread s(a, b)", str(line_spread(parallel.line_a, parallel.line_b)))
            for m_id, m_pt in parallel.meets.items():
                if m_pt is not None:
                    table_par.add_row(f"Meet {m_id}", f"({m_pt.x}, {m_pt.y})")
            m01, m12, m20 = parallel.meets["M01"], parallel.meets["M12"], parallel.meets["M20"]
            if m01 is not None and m12 is not None and m20 is not None:
                table_par.add_row("Quadrance Q(M01, M20)", str(quadrance(m01, m20)))
                table_par.add_row("Quadrance Q(M20, M12)", str(quadrance(m20, m12)))
            table_par.add_row("Pappus Line", f"{parallel.pappus_line.equation()} = 0")
            table_par.add_row("Collinear?", str(parallel.is_collinear))
            console.print(table_par)

            # 3. Homologous Type Triangles (reviving pappus-3.py)
            tri_table = Table(title="Homologous Type Triangles (Square, Circle, Diamond)")
            tri_table.add_column("Type", style="magenta bold")
            tri_table.add_column("Vertices", style="white")
            tri_table.add_column("Exact Area", style="yellow")
            for t_name, poly in general.type_triangles.items():
                v_ids = ", ".join(general.model[v].ID for v in poly.vertices)
                tri_table.add_row(t_name, f"({v_ids})", str(abs(poly.area)))
            console.print(tri_table)

            # 4. Universal algebraic proof over Q(a0, a1, a2, b0, b1, b2, m, c)
            det_expr = verify_pappus_symbolic()
            console.print(
                f"[bold green]Universal symbolic collinearity determinant:[/bold green] {det_expr} (identically zero)"
            )


        if __name__ == "__main__":
            run_demo()

2. demos/pappus_six_permutations.py — Six $S_3$ Lines & Steiner Star Points¶

Constructs all $3! = 6$ permutations on both preset="steiner" and the 2023 preset="parallel" configuration, printing every cross-join meet, Pappus line equation, and exact Star concurrency point $(S_+, S_-)$.

uv run --no-sync python demos/pappus_six_permutations.py
        
r"""Demonstration 2: The Six Permutations (S_3) & Steiner's Two Star Concurrency Points.

        Revives and completes GEOMETOR's 2023 ``pappus.py`` exploration: permuting
        $(B_0, B_1, B_2)$ across all $3! = 6$ elements of $S_3$ yields six Pappus lines
        that split by permutation parity into two concurrent triples:

        - The 3 **even** Pappus lines ($A_3$) concur at the even star point $S_+$.
        - The 3 **odd** Pappus lines ($S_3 \setminus A_3$) concur at the odd star point $S_-$.
        """
        from __future__ import annotations

        import sys
        from pathlib import Path

        _SRC_DIR = Path(__file__).resolve().parents[1] / "src"
        if str(_SRC_DIR) not in sys.path:
            sys.path.insert(0, str(_SRC_DIR))

        from rich.console import Console  # noqa: E402
        from rich.table import Table  # noqa: E402

        from geometor.pappus import (  # noqa: E402
            PERMUTATIONS_S3,
            construct_pappus_configuration,
            permutation_parity,
        )


        def run_demo() -> None:
            """Construct and compare both the 2023 parallel configuration and the compact Steiner configuration."""
            console = Console()

            for preset_key in ("parallel", "steiner"):
                cfg = construct_pappus_configuration(preset=preset_key, scan_harmonics=True)
                table = Table(title=f"Six-Permutation Pappus–Steiner Configuration ({preset_key})")
                table.add_column("#", justify="right", style="dim")
                table.add_column("Perm σ", style="cyan bold")
                table.add_column("Parity", style="magenta")
                table.add_column("Three Cross-Join Meets", style="white")
                table.add_column("Pappus Line Equation (= 0)", style="green")

                for idx, perm in enumerate(PERMUTATIONS_S3):
                    parity = permutation_parity(perm)
                    meets = cfg.meets_by_perm[perm]
                    meet_str = ", ".join(
                        f"{cfg.model[m].ID}=({m.x},{m.y})" if m is not None else "∞"
                        for m in meets
                    )
                    eq_str = str(cfg.pappus_lines[perm].equation())
                    table.add_row(str(idx), str(perm), parity, meet_str, eq_str)
                console.print(table)

                star_table = Table(title=f"Steiner Concurrency Summary ({preset_key})")
                star_table.add_column("Invariant", style="bold yellow")
                star_table.add_column("Exact Symbolic Result", style="cyan")
                star_table.add_row("Total Cross-Join Meets", str(len(cfg.all_meets)))
                if cfg.steiner_even is not None:
                    star_table.add_row("Even Star Point S+ (S1)", f"({cfg.steiner_even.x}, {cfg.steiner_even.y})")
                if cfg.steiner_odd is not None:
                    star_table.add_row("Odd Star Point S- (S2)", f"({cfg.steiner_odd.x}, {cfg.steiner_odd.y})")
                if cfg.steiner_line is not None:
                    star_table.add_row("Steiner Line [S+ S-]", f"{cfg.steiner_line.equation()} = 0")
                star_table.add_row("Collinear Harmonic Tetrads", str(len(cfg.harmonic_ranges)))
                console.print(star_table)


        if __name__ == "__main__":
            run_demo()

3. demos/harmonic_conjugate.py — Complete Quadrilateral & Harmonic Tetrads¶

Verifies Book VII Prop. 129 central perspectivity invariance, demonstrates straightedge fourth harmonic construction across two distinct auxiliary apex/relay choices (confirming invariance of $D = (-4, 0)$), and catalogs emergent harmonic tetrads on preset="harmonic".

uv run --no-sync python demos/harmonic_conjugate.py
        
"""Demonstration 3: Straightedge Fourth Harmonic Conjugate & Projective Invariance.

        Constructs the fourth harmonic conjugate $D$ of $C$ with respect to $(A, B)$
        using only straightedge incidences via a complete quadrilateral, verifies
        $(A, B; C, D) = -1$, and proves that $D$ is completely independent of the
        auxiliary apex $P$ and relay point $Q$.
        """
        from __future__ import annotations

        import sys
        from pathlib import Path

        _SRC_DIR = Path(__file__).resolve().parents[1] / "src"
        if str(_SRC_DIR) not in sys.path:
            sys.path.insert(0, str(_SRC_DIR))

        import sympy as sp  # noqa: E402
        from rich.console import Console  # noqa: E402
        from rich.table import Table  # noqa: E402

        from geometor.pappus import (  # noqa: E402
            construct_harmonic_fourth,
            construct_pappus_configuration,
            cross_ratio,
            harmonic_conjugate_point,
            is_harmonic_range,
        )


        def run_demo() -> None:
            """Run the complete quadrilateral harmonic conjugate construction and invariance check."""
            console = Console()

            res_1 = construct_harmonic_fourth(
                pt_a=(0, 0),
                pt_b=(4, 0),
                pt_c=(sp.Rational(4, 3), 0),
                apex=(2, 3),
                relay=(sp.Rational(5, 3), sp.Rational(3, 2)),
            )
            res_2 = construct_harmonic_fourth(
                pt_a=(0, 0),
                pt_b=(4, 0),
                pt_c=(sp.Rational(4, 3), 0),
                apex=(-1, 5),
                relay=(sp.Rational(2, 5), 2),
            )
            algebraic_d = harmonic_conjugate_point(res_1.pt_a, res_1.pt_b, res_1.pt_c)
            cr_val = cross_ratio(res_1.pt_a, res_1.pt_b, res_1.pt_c, res_1.pt_d)

            q_table = Table(title="Complete Quadrilateral Fourth Harmonic Construction")
            q_table.add_column("Quantity", style="cyan bold")
            q_table.add_column("Exact Value", style="green")
            q_table.add_row("Base Points A, B", f"({res_1.pt_a.x}, {res_1.pt_a.y}), ({res_1.pt_b.x}, {res_1.pt_b.y})")
            q_table.add_row("Interior Point C", f"({res_1.pt_c.x}, {res_1.pt_c.y})")
            q_table.add_row("Constructed D (Trial 1: P=(2,3))", f"({res_1.pt_d.x}, {res_1.pt_d.y})")
            q_table.add_row("Constructed D' (Trial 2: P=(-1,5))", f"({res_2.pt_d.x}, {res_2.pt_d.y})")
            q_table.add_row("Algebraic Formula D", f"({algebraic_d.x}, {algebraic_d.y})")
            q_table.add_row("D == D' == Algebraic?", str(res_1.pt_d.equals(res_2.pt_d) and res_1.pt_d.equals(algebraic_d)))
            q_table.add_row("Cross-Ratio (A, B; C, D)", str(cr_val))
            q_table.add_row("is_harmonic_range(A, B, C, D)", str(is_harmonic_range(res_1.pt_a, res_1.pt_b, res_1.pt_c, res_1.pt_d)))
            console.print(q_table)

            # Emergent harmonic tetrads inside the Pappus configuration
            cfg = construct_pappus_configuration(preset="harmonic", discover_all_intersections=True)
            h_table = Table(title=f"Sample Emergent Harmonic Ranges in Pappus Configuration ({len(cfg.harmonic_ranges)} total)")
            h_table.add_column("#", justify="right", style="dim")
            h_table.add_column("Carrier Line", style="cyan")
            h_table.add_column("Ordered Points (P1 < P2 < P3 < P4)", style="white")
            h_table.add_column("(P1, P3; P2, P4)", style="green")
            for idx, hr in enumerate(cfg.harmonic_ranges[:10], start=1):
                h_table.add_row(str(idx), hr.line_id, f"({', '.join(hr.point_ids)})", str(hr.value))
            console.print(h_table)


        if __name__ == "__main__":
            run_demo()

4. demos/golden_pappus.py — Golden Projectivity in $\mathbb{Q}(\phi)$¶

Computes the canonical golden cross-ratio $(0, 1; \phi, \phi + 1) = \phi$, its six-value anharmonic power orbit, the harmonic-golden identity $(0, \phi; 1, \phi^3) = -1$, and runs geometor.divine to uncover the emergent golden section / A1 M12 B2 /.

uv run --no-sync python demos/golden_pappus.py
        
r"""Demonstration 4: Golden Projectivity, Cross-Ratio Orbits & Book VII Prop. 129.

        Connects Pappus's invariance of the cross-ratio under central perspectivity
        (*Synagoge* Book VII, Prop. 129) to GEOMETOR's central theme of the Golden Ratio
        $\phi = (1 + \sqrt{5})/2$:

        1. Verifies the canonical golden chain cross-ratio $(0, 1; \phi, \phi + 1) = \phi$
           and computes its six-value anharmonic $S_4 / V_4$ orbit in $\mathbb{Q}(\phi)$.
        2. Proves the harmonic-golden identity $(0, \phi; 1, \phi^3) = -1$ where $\phi^3 = 2\phi + 1$.
        3. Projects a golden range onto an oblique line via central perspectivity and
           verifies exact preservation of $\phi$ in the cross-ratio even when Euclidean
           ratios change.
        4. Constructs the Golden Pappus Hexagon over $\mathbb{Q}(\phi)$ and scans it with
           :mod:`geometor.divine`.
        """
        from __future__ import annotations

        import sys
        from pathlib import Path

        _SRC_DIR = Path(__file__).resolve().parents[1] / "src"
        if str(_SRC_DIR) not in sys.path:
            sys.path.insert(0, str(_SRC_DIR))

        import sympy as sp  # noqa: E402
        import sympy.geometry as spg  # noqa: E402
        from geometor.model import Model  # noqa: E402
        from rich.console import Console  # noqa: E402
        from rich.table import Table  # noqa: E402

        from geometor.pappus import (  # noqa: E402
            PHI,
            anharmonic_orbit,
            construct_pappus_hexagon,
            cross_ratio,
            enable_divine_analysis,
            find_golden_sections,
            harmonic_conjugate_point,
            project_point,
        )


        def run_demo() -> None:
            """Execute the Golden Projectivity demonstration."""
            console = Console()

            # 1. Canonical golden chain & anharmonic orbit
            p0 = spg.Point(0, 0)
            p1 = spg.Point(1, 0)
            p2 = spg.Point(PHI, 0)
            p3 = spg.Point(PHI + 1, 0)
            cr_golden = cross_ratio(p0, p1, p2, p3)
            orbit = anharmonic_orbit(cr_golden)

            # 2. Fourth harmonic conjugate of 1 w.r.t. (0, phi) is phi^3 = 2*phi + 1
            h_conj = harmonic_conjugate_point(p0, p2, p1)
            cr_harm_gold = cross_ratio(p0, p2, p1, h_conj)

            # 3. Central perspectivity (Synagoge VII.129) onto an oblique target line
            center = spg.Point(1, 3)
            target_line = spg.Line(spg.Point(0, 1), spg.Point(4, 2))
            projected = [project_point(center, target_line, pt) for pt in (p0, p1, p2, p3)]
            cr_projected = cross_ratio(*projected)

            inv_table = Table(title="Golden Cross-Ratios & Book VII Prop. 129 Perspectivity")
            inv_table.add_column("Property", style="cyan bold")
            inv_table.add_column("Exact Symbolic Result", style="green")
            inv_table.add_row("Golden Ratio φ", f"{PHI} ≈ {float(PHI.evalf()):.10f}")
            inv_table.add_row("Chain Cross-Ratio (0, 1; φ, φ+1)", f"{cr_golden} (equals φ: {sp.simplify(cr_golden - PHI) == 0})")
            inv_table.add_row("6-Value Anharmonic Orbit", ", ".join(str(v) for v in orbit))
            inv_table.add_row("Harmonic Conjugate D of 1 w.r.t. (0, φ)", f"x = {h_conj.x} = 2φ + 1 = φ³ (equals φ³: {sp.simplify(h_conj.x - PHI**3) == 0})")
            inv_table.add_row("Verify (0, φ; 1, φ³)", str(cr_harm_gold))
            inv_table.add_row("Projected Cross-Ratio on x - 4y + 4 = 0", f"{cr_projected} (invariant: {sp.simplify(cr_projected - cr_golden) == 0})")
            console.print(inv_table)

            # 4. Golden Pappus Hexagon with geometor.divine analysis
            model = enable_divine_analysis(Model("golden_pappus"))
            gold_hex = construct_pappus_hexagon(model=model, preset="golden", perm=(0, 1, 2))
            sections = find_golden_sections(gold_hex.model, register_sections=True)

            g_table = Table(title="Golden Pappus Hexagon in Q(φ)")
            g_table.add_column("Element", style="bold yellow")
            g_table.add_column("Exact Symbolic Expression", style="cyan")
            for m_id, m_pt in gold_hex.meets.items():
                if m_pt is not None:
                    g_table.add_row(f"Meet {m_id}", f"({m_pt.x}, {m_pt.y})")
            g_table.add_row("Pappus Line", f"{gold_hex.pappus_line.equation()} = 0")
            g_table.add_row("Collinear over Q(φ)?", str(gold_hex.is_collinear))
            g_table.add_row("Golden Sections Discovered", str(len(sections)))
            for sec in sections:
                g_table.add_row(f"  Golden Section on {sec.line_id}", f"{sec.point_ids} -> ratio = {sec.ratio}")
            console.print(g_table)


        if __name__ == "__main__":
            run_demo()
